    {index grid\\$k\in\{1,\ldots,K\}^{h\times w}$};

\draw[arrow] (encoder) -- (latent);
\draw[arrow] (latent) -- (nearest);
\draw[arrow] (nearest) -- (indices);

\node[below=7mm of sequence, align=center, font=\scriptsize, text width=82mm]
